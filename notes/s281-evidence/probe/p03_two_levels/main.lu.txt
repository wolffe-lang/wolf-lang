enum Kind { Drink, Food, Gratuity }
enum Inner { In(Kind), Nope }
enum Outer { Out(Inner) }

fn deep(o: Outer) -> str {
    match o {
        Out(In(Drink)) => "drink",
        Out(In(_)) => "other",
        Out(Nope) => "nope",
    }
}

fn main() -> !int {
    print(deep(Outer.Out(Inner.In(Kind.Food))))
    print(deep(Outer.Out(Inner.In(Kind.Drink))))
    print(deep(Outer.Out(Inner.Nope)))
    0
}
