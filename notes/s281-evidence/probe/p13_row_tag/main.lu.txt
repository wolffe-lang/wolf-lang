enum Kind { Drink, Food, Gratuity }

fn pick(n: int) -> int ! {bad(Kind)} {
    if n > 0 { n } else if n == 0 { bad(Kind.Food) } else { bad(Kind.Drink) }
}

fn say(n: int) -> str {
    match pick(n) {
        bad(Drink) => "drink",
        bad(_) => "other",
        v => "value {v}",
    }
}

fn main() -> !int {
    print(say(0))
    print(say(-1))
    print(say(5))
    0
}
